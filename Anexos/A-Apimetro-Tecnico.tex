\section{Esquema SQL de Apimetro}
\label{anx:apimetro-sql}

A continuación se presenta el código de creación de las tablas principales
del sistema, que permiten la ingesta asíncrona de datos desde el motor ETL.

\begin{lstlisting}[language=SQL,
    caption={Tablas base de Apimetro: estaciones, líneas e índices espaciales.},
    label=lst:sql-apimetro,
    breaklines=true,
    basicstyle=\small\ttfamily]
-- Tabla de Estaciones (Nodos)
CREATE TABLE estaciones (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(255) NOT NULL,
    sistema VARCHAR(50), -- METRO, MB, RTP, etc.
    alcaldia_municipio VARCHAR(100),
    jerarquia_transporte VARCHAR(50),
    es_cetram BOOLEAN DEFAULT FALSE,
    geom GEOMETRY(Point, 4326) -- SRID WGS 84
);

-- Tabla de Lineas y Ramales (Aristas con metrica)
CREATE TABLE lineas (
    id SERIAL PRIMARY KEY,
    nombre_linea VARCHAR(255),
    sistema VARCHAR(50),
    velocidad_promedio_kmh FLOAT,
    frecuencia_minutos INTEGER,
    capacidad_vehiculo INTEGER,
    sentido INTEGER, -- 1 o 2
    geom GEOMETRY(MultiLineString, 4326)
);

-- Indices espaciales para optimizar consultas de ruteo
CREATE INDEX idx_estaciones_geom ON estaciones USING GIST (geom);
CREATE INDEX idx_lineas_geom ON lineas USING GIST (geom);
\end{lstlisting}

\subsection*{Controlador de estaciones (\texttt{estaciones.go})}

El controlador gestiona el filtrado dinámico de paradas por sistema, alcaldía
y jerarquía de transporte mediante parámetros capturados desde la \textsc{url}.

\begin{lstlisting}[language=Go,
    caption={Lógica de filtrado en \texttt{estaciones.go}.},
    label=lst:go-estaciones]
    // Fragmento simplificado de
    // cmd/pkg/controller/transporte/estaciones.go
    func (ctrl *TransporteController)
        GetEstaciones(c *gin.Context) {
        // Captura de parameters para filtrado dinamico
        sistema := c.Query("sistema")
        numComercial := c.Query("num_comercial")
        alcaldia := c.Query("alcaldia_municipio")
        // Llamada al modelo para
        // ejecucion de query espacial en PostGIS
        estaciones, err := ctrl.model.FetchEstaciones
            (sistema, numComercial, alcaldia)
        if err != nil {
            c.JSON(
                http.StatusInternalServerError,
                gin.H{
                    "error": err.Error()
                    })
            return
        }
        c.JSON(http.StatusOK, estaciones)
    }
\end{lstlisting}
